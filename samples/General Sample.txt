begin
    // Use ONLY integers for everything
    nmbr studentCount = 5
    nmbr total = 0
    nmbr grade
    nmbr i = 1
    
    disp "=== Grade Calculator ==="
    
    for studentCount times
        disp "Enter grade for student " + i + ":"
        input grade
        
        // Simple validation - just make sure it's reasonable
        if grade < 0
            grade = 0
        endif
        
        if grade > 100
            grade = 100
        endif
        
        // Simple grading
        if grade >= 90
            disp "Grade: A"
        elif grade >= 80
            disp "Grade: B"
        elif grade >= 70
            disp "Grade: C"
        elif grade >= 60
            disp "Grade: D"
        else
            disp "Grade: F"
        endif
        
        total = total + grade
        i = i + 1
    stop
    
    nmbr average = total / studentCount
    disp "Average: " + average
    
    // Simple final evaluation
    if average >= 90
        disp "Excellent class! 🌟"
    elif average >= 80
        disp "Good class! 👍"
    elif average >= 70
        disp "Average class. 🙂"
    elif average >= 60
        disp "Below average. 📝"
    else
        disp "Needs improvement. ⚠️"
    endif
close
