begin
    // Basic while loop
    nmbr counter = 1
    
    disp "Counting from 1 to 5:"
    while counter <= 5
        disp "Count: " + counter
        counter = counter + 1
    stop
    
    // While loop with user input
    nmbr total = 0
    nmbr number = 1
    
    disp "Enter numbers to add (enter 0 to stop):"
    
    while number != 0
        input number
        total = total + number
        disp "You entered " + number
    stop
    
    disp "Total sum: " + total
    
    // Password check with while
    string password = "secret123"
    string userInput = ""
    nmbr attempts = 0
    
    while userInput != password && attempts < 3
        disp "Enter password (attempt " + (attempts + 1) + "):"
        input userInput
        attempts = attempts + 1
    stop
    
    if userInput == password
        disp "Access granted! ✅"
    else
        disp "Too many attempts! ❌"
    endif
close
