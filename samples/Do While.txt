begin
    // Basic do-until
    nmbr guess = 0
    nmbr secretNumber = 7
    
    disp "Guess the secret number (1-10):"
    
    do
        disp "Enter your guess:"
        input guess
        disp "You guessed: " + guess
        
        if guess < secretNumber
            disp "Too low! Try again."
        elif guess > secretNumber
            disp "Too high! Try again."
        elif guess == secretNumber
             disp "Congratulations! You guessed it! 🎉"
        endif
    until guess == secretNumber
    
close
